model AIDecisionLog {

#### 2. logDecision() Method Signature

```typescript
async logDecision(data: {
  agentName: string;
  action: string;
  reasoning: string;
  ticketId: string;
  executionId?: string;
  inputData: Record<string, any>;
  outputData: Record<string, any>;
  confidence: number; // 0-1 range
}): Promise<AIDecisionLog | null> {
  try {
    // 1. Create database entry
    const log = await this.prisma.aiDecisionLog.create({
      data: {
        agent_name: data.agentName,
        action: data.action,
        reasoning: data.reasoning,
        ticket_id: data.ticketId,
        execution_id: data.executionId,
        input_data: data.inputData as Prisma.JsonValue,
        output_data: data.outputData as Prisma.JsonValue,
        confidence: data.confidence,
      },
    });

    // 2. Record Prometheus metrics (non-blocking)
    this.metrics.recordCounter('audit_logs_total', { 
      agent_name: data.agentName, 
      action: data.action 
    }).catch(err => this.logger.warn('Failed to record metrics', err));

    // 3. Alert on low confidence (non-blocking)
    if (data.confidence < this.getConfidenceThreshold()) {
      this.metrics.recordCounter('audit_logs_low_confidence_total', { 
        agent_name: data.agentName 
      }).catch(err => this.logger.warn('Failed to record low-confidence metric', err));
      
      this.sentry.captureMessage('Low confidence AI decision', {
        level: 'warning',
        extra: {
          agentName: data.agentName,
          action: data.action,
          confidence: data.confidence,
          ticketId: data.ticketId,
          reasoning: data.reasoning,
        },
      }).catch(err => this.logger.warn('Failed to send Sentry alert', err));
    }

    // 4. Structured logging with correlation ID
    this.logger.log({
      message: `AI decision logged: ${data.agentName} - ${data.action}`,
      confidence: data.confidence,
      ticketId: data.ticketId,
      correlationId: data.executionId, // For cross-service tracing
    });

    return log;
  } catch (error) {
    // Database failure - log but don't crash calling service
    this.logger.error('Failed to create audit log', { 
      error: error.message,
      ticketId: data.ticketId,
      agentName: data.agentName,
    });
    this.sentry.captureException(error);
    return null; // Graceful degradation
  }
}

private getConfidenceThreshold(): number {
  return parseFloat(process.env.AI_CONFIDENCE_THRESHOLD || '0.5');
}
```

#### 3. getDecisionTimeline() Method

**Purpose:** Return AI decisions in chronological order for a specific ticket  
**Prisma Query:** `findMany` with `ticket_id` filter, `orderBy: { timestamp: 'asc' }`, include execution details  
**Use Case:** PM Dashboard (Epic 5) and Lead QA (Epic 8) need timeline view to understand AI reasoning flow

```typescript
async getDecisionTimeline(ticketId: string): Promise<AIDecisionLog[]> {
  return this.prisma.aiDecisionLog.findMany({
    where: { ticket_id: ticketId },
    orderBy: { timestamp: 'asc' },
    include: { execution: { select: { id: true, status: true, provider: true } } },
  });
}
```

#### 4. getLowConfidenceDecisions() Method

**Purpose:** Filter decisions below confidence threshold for human review  
**Prisma Query:** `findMany` with `confidence: { lt: threshold }`, `orderBy: { timestamp: 'desc' }`, `take: 100`  
**Use Case:** Lead QA review queue (Story 8.2) prioritizes manual reviews of low-confidence AI decisions

```typescript
async getLowConfidenceDecisions(threshold: number = 0.5): Promise<AIDecisionLog[]> {
  return this.prisma.aiDecisionLog.findMany({
    where: { confidence: { lt: threshold } },
    orderBy: { timestamp: 'desc' },
    take: 100, // Performance limit
  });
}
```
